Prácticas

Las tres prácticas del curso, desarrolladas paso a paso. Cada una tiene el código completo (también disponible como archivos independientes), la salida esperada y la interpretación que se espera que escribas. Compila siempre con:

g++ -std=c++17 -Wall -Wextra -pthread fichero.cpp -o programa

Práctica 1 · Programación basada en objetos: strings y colecciones

Usar objetos de la biblioteca estándar sin definir clases propias: string, vector, map, flujos.

1.1 a 1.3 · Palabras, ordenación y ficheros

Qué se pide. Leer una línea, imprimir cada palabra en su propia línea sin signos de puntuación, guardarlas en un vector<string>, ordenarlas (de menor a mayor y de mayor a menor) y pasar a leer/escribir ficheros comprobando que entrada y salida son distintos.

Cómo se piensa.

  1. Para separar palabras, un istringstream y in >> p (corta por espacios). Para quitar puntuación, remove_if + erase con ispunct.
  2. sort(v.begin(), v.end()) ordena por código de carácter: las mayúsculas (65-90) van antes que las minúsculas (97-122). Para ignorar mayúsculas, pasa un comparador (lambda) que compare las versiones en minúsculas.
  3. Con ficheros, ifstream y ofstream se usan igual que cin y cout. Comprueba siempre if (!in).
CPP// Práctica 1.1-1.3: palabras de una línea, ordenar y usar ficheros
#include <algorithm>
#include <cctype>
#include <fstream>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

string sin_puntuacion(string p) {
    p.erase(remove_if(p.begin(), p.end(), [](unsigned char c) { return ispunct(c); }), p.end());
    return p;
}
vector<string> palabras(const string& linea) {            // 1.1: separa por espacios con un stringstream
    istringstream in{linea};
    vector<string> v; string p;
    while (in >> p) { p = sin_puntuacion(p); if (!p.empty()) v.push_back(p); }
    return v;
}
string minusculas(string s) { transform(s.begin(), s.end(), s.begin(), [](unsigned char c) { return tolower(c); }); return s; }

int main() {
    string linea = "Programación de Tiempo Real o programación de tiempo real, ¿vale?";
    auto v = palabras(linea);
    cout << "1.1) una por línea:\n"; for (auto& p : v) cout << p << '\n';

    sort(v.begin(), v.end());                              // 1.2: orden por código de carácter
    cout << "1.2) orden por defecto (las mayúsculas van antes que las minúsculas):\n"; for (auto& p : v) cout << p << ' '; cout << '\n';
    sort(v.begin(), v.end(), [](const string& a, const string& b) { return minusculas(a) < minusculas(b); });
    cout << "   ignorando mayúsculas:\n"; for (auto& p : v) cout << p << ' '; cout << '\n';
    sort(v.begin(), v.end(), [](const string& a, const string& b) { return minusculas(a) > minusculas(b); });
    cout << "   de mayor a menor:\n"; for (auto& p : v) cout << p << ' '; cout << '\n';

    // 1.3: ficheros (comprobando que entrada y salida son distintos)
    string entrada = "entrada.txt", salida = "salida.txt";
    if (entrada == salida) { cerr << "los ficheros deben ser distintos\n"; return 1; }
    { ofstream f{entrada}; f << linea << '\n'; }           // preparo un fichero de prueba
    ifstream in{entrada}; ofstream out{salida};
    if (!in || !out) { cerr << "no se pudo abrir\n"; return 1; }
    string l;
    while (getline(in, l)) for (auto& p : palabras(l)) out << p << '\n';
    cout << "1.3) escrito " << salida << '\n';
}

1.4 y 1.5 · Histograma de palabras

Qué se pide. Un map<string,int> con el número de apariciones de cada palabra; mostrar las dos más repetidas y qué palabras acumulan el 20 % de las apariciones; después, un histograma por longitud con vector<map<string,int>>.

Cómo se piensa.

  • ++hist[palabra] hace todo: si no existe la crea con 0 y suma 1.
  • Un map está ordenado por clave, no por valor. Para ordenar por frecuencia, copia a un vector<pair<string,int>> y ordénalo con un comparador por second.
  • El 20 %: recorre ordenado de más a menos frecuente acumulando hasta llegar a total/5.
  • Por longitud: por_longitud[palabra.size()][palabra].
CPP// Práctica 1: histograma de palabras de un fichero (map<string,int>) y por longitud (vector<map>)
#include <algorithm>
#include <cctype>
#include <fstream>
#include <iostream>
#include <map>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

string normalizar(string p) {
    p.erase(remove_if(p.begin(), p.end(), [](unsigned char c) { return ispunct(c); }), p.end());   // sin puntuación
    transform(p.begin(), p.end(), p.begin(), [](unsigned char c) { return tolower(c); });         // sin mayúsculas (solo ASCII)
    return p;
}

int main(int argc, char* argv[]) {
    if (argc != 2) { cerr << "uso: " << argv[0] << " fichero.txt\n"; return 1; }
    ifstream in{argv[1]};
    if (!in) { cerr << "no se puede abrir " << argv[1] << '\n'; return 1; }

    map<string, int> hist;
    vector<map<string, int>> por_longitud(16);         // índice = nº de letras (suponemos <= 15)
    long total = 0;
    string palabra;
    while (in >> palabra) {                            // >> separa por espacios
        palabra = normalizar(palabra);
        if (palabra.empty()) continue;
        ++hist[palabra]; ++total;
        if (palabra.size() < por_longitud.size()) ++por_longitud[palabra.size()][palabra];
    }
    // ordenar por frecuencia: un map está ordenado por clave, así que pasamos a un vector de pares
    vector<pair<string, int>> v(hist.begin(), hist.end());
    sort(v.begin(), v.end(), [](const auto& a, const auto& b) { return a.second > b.second || (a.second == b.second && a.first < b.first); });

    cout << "palabras: " << total << ", distintas: " << hist.size() << '\n';
    cout << "las dos mas repetidas:\n";
    for (size_t i = 0; i < 2 && i < v.size(); ++i) cout << "  " << v[i].first << ": " << v[i].second << '\n';
    long acum = 0; cout << "palabras que acumulan el 20% de las apariciones:";
    for (auto& [p, n] : v) { cout << ' ' << p << '(' << n << ')'; acum += n; if (acum * 5 >= total) break; }
    cout << "\nla mas repetida de cada longitud (1-6 letras):\n";
    for (size_t l = 1; l <= 6; ++l) {
        if (por_longitud[l].empty()) continue;
        auto it = max_element(por_longitud[l].begin(), por_longitud[l].end(), [](auto& a, auto& b) { return a.second < b.second; });
        cout << "  " << l << " letras: " << it->first << " (" << it->second << ")\n";
    }
}

Interpretación. Con el fragmento de 1882 palabras, de (123) e y (104) suman casi el 12 % del texto: las palabras funcionales dominan. Cinco palabras (de, y, que, a, el) acumulan el 20 %. Es la ley de Zipf: la frecuencia cae muy deprisa con el puesto en el ranking.

Qué suele fallar
  • hist[palabra] para consultar si existe: la crea. Usa find.
  • Olvidar convertir a minúsculas: El y el cuentan como palabras distintas.
  • No comprobar el fichero abierto, o dejar argv[1] sin comprobar argc.
  • Los acentos: tolower del C solo funciona con ASCII. É no se convierte.

Práctica 2 · Programación orientada a objetos

2.1 · Clase número complejo

Qué se pide. Atributos real e imaginario; constructor con dos valores y un constructor delegado sin parámetros (0+0i); get; módulo y argumento como pair con hypot y atan2; operator<<, operator>>, +, -.

Decisiones de diseño. Los operadores y modulo_argumento pueden ser funciones libres (fuera de la clase) porque solo necesitan los get: la clase queda más pequeña y estable. operator>> construye un temporal y hace swap: si la lectura falla, el objeto original no queda a medias.

CPP// Práctica 2.1: número complejo
#include <cmath>
#include <iostream>
#include <utility>
using namespace std;

class C_Complejo {
public:
    C_Complejo(double re, double im) : re_{re}, im_{im} {}
    C_Complejo() : C_Complejo(0, 0) {}                     // constructor delegado: 0 + 0i
    double real() const { return re_; }
    double imag() const { return im_; }
private:
    double re_, im_;
};

// Fuera de la clase: solo necesita los get. Reduce lo que la clase tiene que mantener.
pair<double, double> modulo_argumento(const C_Complejo& c) { return {hypot(c.real(), c.imag()), atan2(c.imag(), c.real())}; }

C_Complejo operator+(const C_Complejo& l, const C_Complejo& r) { return {l.real() + r.real(), l.imag() + r.imag()}; }
C_Complejo operator-(const C_Complejo& l, const C_Complejo& r) { return {l.real() - r.real(), l.imag() - r.imag()}; }
ostream& operator<<(ostream& os, const C_Complejo& c) { return os << c.real() << (c.imag() < 0 ? "-" : "+") << fabs(c.imag()) << "i"; }
istream& operator>>(istream& is, C_Complejo& c) {            // formato: re im
    double re, im;
    if (is >> re >> im) { C_Complejo tmp{re, im}; swap(c, tmp); }   // construir temporal y swap
    return is;
}

int main() {
    C_Complejo a{3, 4}, b{1, -2}, z;
    cout << "a=" << a << " b=" << b << " z=" << z << '\n';
    cout << "a+b=" << a + b << "  a-b=" << a - b << '\n';
    auto [m, arg] = modulo_argumento(a);
    cout << "|a|=" << m << " arg(a)=" << arg << " rad\n";
}

a+b=4+2i  a-b=2+6i
|a|=5 arg(a)=0.927295 rad

2.2 y 2.3 · Mensajes secretos y su registro

Qué se pide. Mensaje_Secreto con texto, clave y pista; bloquear (sin y con clave nueva, esta última solo si está desbloqueado), desbloquear(clave), esta_bloqueado, borrar; operator<< que muestra la pista si está bloqueado. Registro_Mensajes con un map<string, vector<Mensaje_Secreto>> y métodos para añadir, consultar, leer el primero/último (borrándolo), listar y borrar con claves.

Cómo se piensa.

  • total_mensajes_persona usa find (no []) para no crear personas fantasma y convierte size() (sin signo) a int.
  • primer_mensaje y ultimo_mensaje se parecen mucho: un método privado leer(persona, clave, primero) evita repetir código.
  • Borrar del vector mientras se recorre exige iteradores: m = erase(m) o ++m.
CPP// Prácticas 2.2 y 2.3: Mensaje_Secreto y Registro_Mensajes
#include <iostream>
#include <map>
#include <string>
#include <vector>
using namespace std;

class Mensaje_Secreto {
public:
    Mensaje_Secreto(string texto, int clave, string pista = "") : texto_{std::move(texto)}, clave_{clave}, pista_{std::move(pista)} {}
    void bloquear() { bloqueado_ = true; }
    bool bloquear(int clave_nueva) { if (bloqueado_) return false; clave_ = clave_nueva; bloqueado_ = true; return true; }
    bool desbloquear(int clave) { if (clave == clave_) bloqueado_ = false; return !bloqueado_; }
    bool esta_bloqueado() const { return bloqueado_; }
    void borrar() { texto_.clear(); }
    const string& pista() const { return pista_; }
    friend ostream& operator<<(ostream& os, const Mensaje_Secreto& m) {
        return os << (m.bloqueado_ ? "[bloqueado] pista: " + m.pista_ : m.texto_);
    }
private:
    string texto_; int clave_; string pista_; bool bloqueado_ = true;
};

class Registro_Mensajes {
public:
    void nuevo_mensaje(const Mensaje_Secreto& m, const string& persona) { reg_[persona].push_back(m); }
    bool existe_persona(const string& p) const { return reg_.find(p) != reg_.end(); }
    int total_mensajes_persona(const string& p) const { auto it = reg_.find(p); return it == reg_.end() ? 0 : static_cast<int>(it->second.size()); }
    vector<string> lista_personas() const { vector<string> v; for (const auto& [p, _] : reg_) v.push_back(p); return v; }
    bool primer_mensaje(const string& p, int clave) { return leer(p, clave, true); }
    bool ultimo_mensaje(const string& p, int clave) { return leer(p, clave, false); }
    int borrar_todos(const string& p, const vector<int>& claves) {      // claves probadas una por mensaje
        auto it = reg_.find(p); if (it == reg_.end()) return 0;
        int borrados = 0; size_t k = 0;
        for (auto m = it->second.begin(); m != it->second.end(); ++k) {
            if (k < claves.size() && m->desbloquear(claves[k])) { m->borrar(); m = it->second.erase(m); ++borrados; }
            else ++m;                                                    // clave incorrecta: se queda
        }
        return borrados;
    }
private:
    bool leer(const string& p, int clave, bool primero) {                // método privado común
        auto it = reg_.find(p);
        if (it == reg_.end() || it->second.empty()) { cout << "sin mensajes para " << p << '\n'; return false; }
        auto m = primero ? it->second.begin() : prev(it->second.end());
        cout << "pista: " << m->pista() << '\n';
        if (!m->desbloquear(clave)) { cout << "clave incorrecta\n"; return false; }
        cout << *m << '\n'; m->borrar(); it->second.erase(m);
        return true;
    }
    map<string, vector<Mensaje_Secreto>> reg_;
};

int main() {
    Registro_Mensajes r;
    r.nuevo_mensaje({"hola Ana", 1234, "mi PIN"}, "Ana");
    r.nuevo_mensaje({"adios Ana", 99, "dos cifras"}, "Ana");
    cout << "mensajes de Ana: " << r.total_mensajes_persona("Ana") << ", de Luis: " << r.total_mensajes_persona("Luis") << '\n';
    r.primer_mensaje("Ana", 1);        // clave incorrecta
    r.primer_mensaje("Ana", 1234);     // lo muestra y lo borra
    r.ultimo_mensaje("Ana", 99);
    r.primer_mensaje("Luis", 0);
    cout << "quedan " << r.total_mensajes_persona("Ana") << " mensajes\n";
}

2.5 · Actividades de un proyecto

Misma idea con un contenedor recursivo, ahora con dos niveles de abstracción: Tarea_Compuesta es abstracta (no implementa duracion) y sus dos hijas deciden cómo combinarla. Paralelas: el máximo de las duraciones. Secuenciales: la suma.

CPP// Práctica 2.5: actividades de un proyecto (contenedor recursivo con herencia)
#include <algorithm>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

class Actividad {
public:
    Actividad(string id, string desc) : id_{std::move(id)}, desc_{std::move(desc)} {}
    virtual ~Actividad() = default;
    virtual int duracion() const = 0;
    friend ostream& operator<<(ostream& os, const Actividad& a) { return os << a.to_string(); }
protected:
    virtual string to_string() const { ostringstream o; o << id_ << " (" << desc_ << ") dura " << duracion(); return o.str(); }
    string id_, desc_;
};

class Tarea : public Actividad {
public:
    Tarea(string id, string desc, int d) : Actividad{std::move(id), std::move(desc)}, d_{d} {}
    int duracion() const override { return d_; }
private:
    int d_;
};

class Tarea_Compuesta : public Actividad {        // abstracta: no implementa duracion()
public:
    Tarea_Compuesta(string id, string desc) : Actividad{std::move(id), std::move(desc)} {}
    bool anyadir(Actividad* a) { if (a == this) return false; v_.push_back(a); return true; }
protected:
    string to_string() const override {
        ostringstream o; o << Actividad::to_string() << " {";
        for (auto* a : v_) o << " [" << *a << "]";
        o << " }"; return o.str();
    }
    vector<Actividad*> v_;
};

class Tareas_Paralelas : public Tarea_Compuesta {
public:
    using Tarea_Compuesta::Tarea_Compuesta;       // hereda constructores
    int duracion() const override { int m = 0; for (auto* a : v_) m = max(m, a->duracion()); return m; }   // el máximo
};
class Tareas_Secuenciales : public Tarea_Compuesta {
public:
    using Tarea_Compuesta::Tarea_Compuesta;
    int duracion() const override { int s = 0; for (auto* a : v_) s += a->duracion(); return s; }          // la suma
};

int main() {
    Tarea t1{"T1", "cavar", 3}, t2{"T2", "cimentar", 5}, t3{"T3", "pintar", 2}, t4{"T4", "limpiar", 1};
    Tareas_Paralelas par{"P", "a la vez"};
    par.anyadir(&t1); par.anyadir(&t2);
    Tareas_Secuenciales seq{"S", "en orden"};
    seq.anyadir(&par); seq.anyadir(&t3); seq.anyadir(&t4);
    cout << seq << '\n';
    cout << "duracion total = " << seq.duracion() << " (max(3,5) + 2 + 1)\n";
}
SalidaS (en orden) dura 8 { [P (a la vez) dura 5 { [T1 (cavar) dura 3] [T2 (cimentar) dura 5] }] [T3 (pintar) dura 2] [T4 (limpiar) dura 1] }
duracion total = 8 (max(3,5) + 2 + 1)

Interpretación. La duración de S es 8 = max(3, 5) + 2 + 1: la tarea paralela interna aporta 5 (la más larga), no 3 + 5. El polimorfismo hace que S.duracion() llame a P.duracion() sin saber qué tipo de tarea es.

Práctica 3 · Programación concurrente

3.1 y 3.3 · Suma de matrices con hilos y con thread pool

Qué se pide. Sumar dos matrices vector<vector<int>> con \(n\) hilos de forma que no haya carreras sin mutex: cada hilo suma filas disjuntas. Después, un thread pool: los hilos piden al «director» la siguiente fila.

Cómo se piensa.

  • 3.1: el hilo \(id\) suma las filas \(id, id+n, id+2n, \ldots\). Como cada hilo escribe en filas distintas de c, no hay carrera y no hace falta mutex. Las matrices de entrada solo se leen: se pasan con cref.
  • 3.3: ahora el reparto es dinámico. El contador de «siguiente fila» sí es un dato compartido: lo protege un mutex dentro de la clase Director.
CPP// Prácticas 3.1 y 3.3: suma de matrices por filas con n hilos y con thread pool
#include <chrono>
#include <iostream>
#include <mutex>
#include <thread>
#include <vector>
using namespace std;
using Matriz = vector<vector<int>>;

Matriz crear(int tam) {                                   // valores 0..tam-1 en cada fila
    Matriz m(tam, vector<int>(tam));
    for (auto& fila : m) for (int j = 0; j < tam; ++j) fila[j] = j;
    return m;
}
ostream& operator<<(ostream& os, const Matriz& m) {
    for (const auto& f : m) { for (int x : f) os << x << ' '; os << '\n'; }
    return os;
}
// 3.1: cada hilo suma las filas inicio, inicio+salto, inicio+2*salto... No comparten escritura: no hace falta mutex.
void suma_filas(const Matriz& a, const Matriz& b, Matriz& c, size_t inicio, size_t salto) {
    for (size_t i = inicio; i < a.size(); i += salto)
        for (size_t j = 0; j < a[i].size(); ++j) c[i][j] = a[i][j] + b[i][j];
}
Matriz suma_paralelo(const Matriz& a, const Matriz& b, int n_hilos) {
    Matriz c(a.size(), vector<int>(a.size()));
    vector<thread> hilos;
    for (int id = 0; id < n_hilos; ++id) hilos.emplace_back(suma_filas, cref(a), cref(b), ref(c), id, n_hilos);
    for (auto& h : hilos) h.join();
    return c;
}
// 3.3: el «director» reparte la siguiente fila a quien la pida. Su contador sí es compartido: mutex.
class Director {
public:
    explicit Director(size_t n_filas) : n_{n_filas} {}
    bool siguiente(size_t& fila) { lock_guard<mutex> lk{m_}; if (sig_ >= n_) return false; fila = sig_++; return true; }
private:
    mutex m_; size_t sig_ = 0, n_;
};
Matriz suma_pool(const Matriz& a, const Matriz& b, int n_hilos) {
    Matriz c(a.size(), vector<int>(a.size()));
    Director dir{a.size()};
    auto trabajador = [&] {
        size_t f;
        while (dir.siguiente(f))
            for (size_t j = 0; j < a[f].size(); ++j) c[f][j] = a[f][j] + b[f][j];
    };
    vector<thread> hilos;
    for (int i = 0; i < n_hilos; ++i) hilos.emplace_back(trabajador);
    for (auto& h : hilos) h.join();
    return c;
}

int main() {
    Matriz a = crear(4), b = crear(4);
    cout << "a + b con 3 hilos:\n" << suma_paralelo(a, b, 3);
    cout << "mismo resultado con thread pool: " << boolalpha << (suma_pool(a, b, 3) == suma_paralelo(a, b, 3)) << '\n';

    const int N = 1500;
    Matriz A = crear(N), B = crear(N);
    for (int nh : {1, 2, 4}) {
        auto t0 = chrono::steady_clock::now();
        Matriz C = suma_paralelo(A, B, nh);
        auto t1 = chrono::steady_clock::now();
        cout << nh << " hilos: " << chrono::duration_cast<chrono::microseconds>(t1 - t0).count() << " us (C[7][7]=" << C[7][7] << ")\n";
    }
}

Interpretación. La mejora real es modesta: sumar es muy barato y el tiempo lo domina el acceso a memoria y la reserva de la matriz resultado. Más hilos que núcleos no ayuda. Para un cálculo más costoso por elemento, la mejora sí se acerca al número de núcleos.

3.2 · La cena de los filósofos

Qué se pide. \(n\) filósofos, \(n+1\) cubiertos (un mutex cada uno). Para comer, el filósofo \(i\) necesita tres cubiertos: \(i\), \(i+1\), \(i+2\) (módulo \(n+1\)). Ciclo: coge cubiertos, come 10 ms, los suelta, piensa 10 ms. Usar std::lock o scoped_lock; probar después con lock() individuales y observar el interbloqueo.

Cómo se piensa. Con lock() uno a uno, cada filósofo se queda con su primer cubierto y espera el segundo, que tiene otro: espera circular ⇒ interbloqueo. scoped_lock lk{c1, c2, c3} los coge todos a la vez con un algoritmo seguro.

CPP// Práctica 3.2: cena de los filósofos con 3 cubiertos por filósofo (n filósofos, n+1 cubiertos)
#include <chrono>
#include <iostream>
#include <mutex>
#include <thread>
#include <vector>
using namespace std;
using namespace std::chrono_literals;

mutex consola;
void imprimir(const string& s) { lock_guard<mutex> lk{consola}; cout << s << '\n'; }

void filosofo(int id, mutex& c1, mutex& c2, mutex& c3, int ciclos) {
    for (int i = 0; i < ciclos; ++i) {
#ifdef INTERBLOQUEO
        c1.lock(); this_thread::sleep_for(1ms); c2.lock(); c3.lock();     // uno a uno: cada hilo retiene y espera
#else
        scoped_lock lk{c1, c2, c3};                                       // los tres a la vez, sin interbloqueo
#endif
        imprimir("F" + to_string(id) + " come");
        this_thread::sleep_for(10ms);
#ifdef INTERBLOQUEO
        c3.unlock(); c2.unlock(); c1.unlock();
#endif
        imprimir("F" + to_string(id) + " piensa");
        this_thread::sleep_for(10ms);
    }
}

int main(int argc, char* argv[]) {
    int n = argc > 1 ? stoi(argv[1]) : 3, ciclos = argc > 2 ? stoi(argv[2]) : 2;
    vector<mutex> cubiertos(n + 1);                      // un mutex por cubierto
    vector<thread> hilos;
    for (int i = 0; i < n; ++i)                           // F_i usa los cubiertos i, i+1, i+2 (módulo n+1)
        hilos.emplace_back(filosofo, i + 1, ref(cubiertos[i]), ref(cubiertos[(i + 1) % (n + 1)]), ref(cubiertos[(i + 2) % (n + 1)]), ciclos);
    for (auto& h : hilos) h.join();
    imprimir("todos han terminado");
}

Con -DINTERBLOQUEO (cada lock() por separado y una pausa de 1 ms entre ellos) el programa se bloquea: en cinco ejecuciones de prueba con 3 filósofos y 50 ciclos se cortó por tiempo en las cinco. Sin esa definición, termina siempre:

OUTF1 come
F2 piensa
...
todos han terminado

3.4 · Almacén (monitor)

Qué se pide. Un generador que escribe los valores 0 a 6 y varios lectores. Cada lector debe leer cada valor una vez y el generador no puede escribir el siguiente hasta que todos hayan leído el actual. Clase Almacen como monitor, con cierre ordenado.

Cómo se piensa. Estado: el valor y un map<thread::id,bool> que dice quién ya lo leyó. Dos condiciones de espera, ambas sobre el mismo mutex:

  • El generador espera hasta que todos_han_leido().
  • Un lector espera hasta que él no ha leído el valor actual.

Cada cambio de estado (anyadir, leer) hace notify_all: así se despierta quien corresponda y los demás recomprueban su predicado. cerrar() pone una bandera y despierta a todos para que ninguno se quede dormido para siempre.

CPP// Práctica 3.4: Almacen (monitor). Un generador y varios lectores: cada lector lee cada valor UNA vez,
// y el generador no escribe el siguiente hasta que todos los lectores han leído el actual.
#include <atomic>
#include <chrono>
#include <condition_variable>
#include <iostream>
#include <map>
#include <mutex>
#include <thread>
#include <vector>
using namespace std;
using namespace std::chrono_literals;

class Almacen {
public:
    void anyadir_lector(thread::id id) { lock_guard<mutex> lk{m_}; leido_[id] = true; }     // empieza «sin pendiente»
    void anyadir(int v) {
        unique_lock<mutex> lk{m_};
        cv_.wait(lk, [&] { return todos_han_leido() || cerrado_; });                         // espera si alguien no ha leído
        if (cerrado_) return;
        valor_ = v;
        for (auto& [id, ok] : leido_) ok = false;                                            // ahora todos tienen pendiente
        cv_.notify_all();
    }
    bool leer(int& v) {                                                                      // false = almacén cerrado
        unique_lock<mutex> lk{m_};
        auto yo = this_thread::get_id();
        cv_.wait(lk, [&] { return !leido_[yo] || cerrado_; });                               // espera un valor nuevo
        if (leido_[yo]) return false;                                                        // cerrado y sin pendiente
        v = valor_; leido_[yo] = true;
        cv_.notify_all();                                                                    // quizá era el último lector
        return true;
    }
    void cerrar() { { lock_guard<mutex> lk{m_}; cerrado_ = true; } cv_.notify_all(); }
private:
    bool todos_han_leido() const { for (const auto& [id, ok] : leido_) if (!ok) return false; return true; }
    mutex m_;
    condition_variable cv_;
    map<thread::id, bool> leido_;
    int valor_ = 0;
    bool cerrado_ = false;
};

int main() {
    Almacen a;
    atomic<bool> seguir{true};
    vector<long> suma(3, 0), cuenta(3, 0);
    vector<thread> lectores;
    for (int i = 0; i < 3; ++i)
        lectores.emplace_back([&, i] {
            a.anyadir_lector(this_thread::get_id());
            int v;
            while (seguir && a.leer(v)) { suma[i] += v; ++cuenta[i]; this_thread::sleep_for(chrono::milliseconds(2 + 3 * i)); }
        });
    this_thread::sleep_for(50ms);                                  // deja que se registren los lectores
    thread generador{[&] { for (int v = 0; v <= 6; ++v) a.anyadir(v); }};
    generador.join();
    this_thread::sleep_for(60ms);                                  // los lectores terminan de leer el último valor
    seguir = false; a.cerrar();
    for (auto& h : lectores) h.join();
    for (int i = 0; i < 3; ++i) cout << "lector " << i << ": " << cuenta[i] << " valores, suma " << suma[i] << '\n';
    cout << "(cada lector debe leer 7 valores y sumar 21)\n";
}
Salidalector 0: 7 valores, suma 21
lector 1: 7 valores, suma 21
lector 2: 7 valores, suma 21
(cada lector debe leer 7 valores y sumar 21)

Interpretación. Cada lector lee exactamente 7 valores y suma 21 (0+1+…+6), sea cual sea la velocidad de cada uno: la sincronización fuerza el ritmo del más lento. Es el esquema de «productor con varios consumidores que necesitan todos el mismo dato».

Resumen de prácticas
  • P1: ++hist[p], find para consultar, sort con lambda para ordenar por valor.
  • P2: constructor delegado, operator<< con to_string() virtual, destructor virtual, contenedor recursivo con punteros.
  • P3: filas disjuntas ⇒ sin mutex; contador compartido ⇒ mutex; varios recursos ⇒ scoped_lock; esperar condición ⇒ condition_variable con predicado.
